# LinkedIn Personal Profile Scraping Instructions

# Loop through each profile URL in the provided file
FOR_EACH_URL IN ${URLS_FILE}
    # 1. Visit the profile
    VISIT_URL

    # 2. Wait for page load and basic elements
    WAIT 5
    WAIT_FOR "//h1[contains(@class, 't-24')]"

    # 3. Extract basic profile information
    EXTRACT {"name": "//h1[contains(@class, 't-24')]", "headline": "//div[contains(@class, 'text-body-medium')]", "about": "//div[@id='about']//p"}

    # 4. Click Contact Info button and wait
    CLICK_ELEMENT_BY_SELECTOR "//a[contains(@id, 'contact-info')]"
    WAIT 2
    WAIT_FOR "//div[@role='dialog']"

    # 5. Extract contact information from dialog
    EXTRACT {"linkedin_url": "//div[@role='dialog']//a[contains(@href, 'linkedin.com/in/')]/@href", "websites": "//div[@role='dialog']//section[.//h3[text()='Website']]//a/@href", "phone": "//div[@role='dialog']//section[.//h3[text()='Phone']]//span", "email": "//div[@role='dialog']//section[.//h3[text()='Email']]//a", "birthday": "//div[@role='dialog']//section[.//h3[text()='Birthday']]//span"}

    # 6. Save extracted data (this will save both basic profile and contact info)
    SAVE_TO ${OUTPUT_FILE}

    # 7. Wait between profiles to avoid rate limiting
    WAIT_RANDOM 10-15
END_FOR

# Mark scraping as complete
MARK_COMPLETE
