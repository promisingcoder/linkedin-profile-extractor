# LinkedIn Login Instructions
# Navigate to LinkedIn homepage
NAVIGATE https://www.linkedin.com/

# Wait for sign in button and click it
WAIT_FOR //a[@data-test-id='home-hero-sign-in-cta']
CLICK_ELEMENT_BY_SELECTOR //a[@data-test-id='home-hero-sign-in-cta']

# Wait longer for login form to be fully loaded
WAIT 5

# Fill username
WAIT_FOR //input[@id='username']
WAIT 2
FILL_INPUT "//input[@id='username']" ${LINKEDIN_EMAIL}
WAIT 2

# Fill password
WAIT_FOR //input[@id='password']
WAIT 2
FILL_INPUT "//input[@id='password']" ${LINKEDIN_PASSWORD}
WAIT 2

# Submit login form
WAIT_FOR //button[@type='submit']
WAIT 2
CLICK_ELEMENT_BY_SELECTOR //button[@type='submit']

# Wait for successful login (wait for nav menu to load)
WAIT_FOR //nav[contains(@class, 'global-nav')]

# Save cookies after successful login
SAVE_TO_FILE ${COOKIES_FILE}
